#include <iostream>
#include <curses.h>
#include <cstring>
#include <ctime>
#include <cstdlib>

using namespace std;

const char CLASSIFICA_FILE[] = "classifica.txt";
const int MAX_CLASSIFICA = 5;

void mostraClassifica() {
    FILE *file = fopen(CLASSIFICA_FILE, "r");
    clear();
    mvprintw(2, 2, "Classifica:");
    int y = 4;

    if (file) {
        int punteggio;
        char nome[20];
        int i = 1;
        while (fscanf(file, "%19s %d", nome, &punteggio) == 2) {
            mvprintw(y++, 4, "%d) %s - %d", i++, nome, punteggio);
        }
        fclose(file);
    }

    mvprintw(y + 2, 2, "Premi un tasto per tornare al menu...");
    refresh();
    getch();
}

void aggiornaClassifica(int nuovoPunteggio, const char *nome) {
    FILE *file = fopen(CLASSIFICA_FILE, "r");
    int punteggi[MAX_CLASSIFICA + 1] = {0};
    char nomi[MAX_CLASSIFICA + 1][20] = {{0}};
    int count = 0;

    if (file) {
        while (fscanf(file, "%19s %d", nomi[count], &punteggi[count]) == 2) {
            count++;
        }
        fclose(file);
    }

    punteggi[count] = nuovoPunteggio;
    strncpy(nomi[count], nome, 19);
    count++;

    for (int i = 0; i < count - 1; i++) {
        for (int j = i + 1; j < count; j++) {
            if (punteggi[j] > punteggi[i]) {
                int tempP = punteggi[i];
                punteggi[i] = punteggi[j];
                punteggi[j] = tempP;

                char tempN[20];
                strcpy(tempN, nomi[i]);
                strcpy(nomi[i], nomi[j]);
                strcpy(nomi[j], tempN);
            }
        }
    }

    file = fopen(CLASSIFICA_FILE, "w");
    for (int i = 0; i < (count < MAX_CLASSIFICA ? count : MAX_CLASSIFICA); i++) {
        fprintf(file, "%s %d\n", nomi[i], punteggi[i]);
    }
    fclose(file);
}

void cancellaClassifica() {
    system("echo -n > classifica.txt");
}

void menu() {
    int evidenziato = 0;
    const char *scelte[] = { "Nuova Partita", "Visualizza Classifica", "Cancella Classifica", "Esci" };
    int scelta;
    initscr();
    noecho();
    curs_set(0);
    keypad(stdscr, TRUE);

    while (true) {
        clear();
        mvprintw(2, 2, "Menu");

        for (int i = 0; i < 4; i++) {
            if (i == evidenziato) {
                attron(A_REVERSE);
            }
            mvprintw(4 + i, 4, "%s", scelte[i]);
            if (i == evidenziato) {
                attroff(A_REVERSE);
            }
        }
        refresh();

        scelta = getch();

        switch (scelta) {
            case KEY_UP:
                evidenziato = (evidenziato == 0) ? 3 : evidenziato - 1;
                break;
            case KEY_DOWN:
                evidenziato = (evidenziato == 3) ? 0 : evidenziato + 1;
                break;
            case 10: // INVIO
                if (evidenziato == 0) {
                    clear();
                    mvprintw(2, 2, "Inserisci il tuo nome: ");
                    char nome[20];
                    echo();
                    curs_set(1);
                    getnstr(nome, 19);
                    noecho();
                    curs_set(0);
                    int punteggio = rand() % 1000; // Simula punteggio casuale
                    aggiornaClassifica(punteggio, nome);
                } else if (evidenziato == 1) {
                    mostraClassifica();
                } else if (evidenziato == 2) {
                    cancellaClassifica();
                } else {
                    endwin();
                    return;
                }
                break;
            case 'T':
            case 't':
                endwin();
                return;
        }
    }
    endwin();
}

int main() {
    srand(time(nullptr));
    menu();
    return 0;
}
